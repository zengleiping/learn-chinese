/* 連連看：即時同步雙方選取狀態與醒目錯誤紅框 */ function renderMatchBoard(s, container){ var progress = getProgress(s.id); var matched = progress.matched || {}; var selections = progress.selections || {}; var interactive = canInteract(); var wrap = document.createElement("div"); wrap.className = "match-board"; var svg = document.createElementNS("http://www.w3.org/2000/svg","svg"); svg.setAttribute("class","match-svg"); wrap.appendChild(svg); var cols = document.createElement("div"); cols.className = "match-cols"; var lCol = document.createElement("div"); lCol.className = "match-col"; var rCol = document.createElement("div"); rCol.className = "match-col"; lCol.innerHTML = '
' + (s.leftLabel || "項目一") + '
'; rCol.innerHTML = '
' + (s.rightLabel || "項目二") + '
'; var pIds = Object.keys(s.pairs).sort(); var rOrder = pIds.slice().sort(); var selL = null, selR = null; pIds.forEach(function(pid){ var p = s.pairs[pid]; var btn = document.createElement("button"); btn.type="button"; var isMatched = !!matched[pid]; btn.className = "match-item zh" + (isMatched ? " matched" : ""); btn.textContent = p.left; btn.dataset.id = pid; btn.dataset.side = "left"; // 即時檢查是否有任何連線者正在點選此項 var userWhoSelectedL = Object.keys(selections).find(function(uid){ return selections[uid] && selections[uid].left === pid; }); if(userWhoSelectedL && !isMatched){ btn.classList.add("selected"); var uInfoL = selections[userWhoSelectedL]; var uColorL = getColorForStudent(uInfoL.name || "學生", uInfoL.role); btn.style.borderColor = uColorL; btn.style.boxShadow = "0 0 0 3.5px " + uColorL + "55 inset"; } else if(!isMatched){ btn.style.boxShadow = ""; } if(isMatched || !interactive) btn.disabled = true; btn.onclick = function(){ if(matched[pid] || !interactive) return; selL = pid; var mySel = { left: selL, right: selR, name: myName, role: roomRole }; var newSelections = Object.assign({}, selections, { [myId]: mySel }); writeProgress(s.id, { selections: newSelections }); checkMatch(); }; lCol.appendChild(btn); }); rOrder.forEach(function(pid){ var p = s.pairs[pid]; var btn = document.createElement("button"); btn.type="button"; var isMatched = !!matched[pid]; btn.className = "match-item" + (isMatched ? " matched" : ""); btn.textContent = p.right; btn.dataset.id = pid; btn.dataset.side = "right"; var userWhoSelectedR = Object.keys(selections).find(function(uid){ return selections[uid] && selections[uid].right === pid; }); if(userWhoSelectedR && !isMatched){ btn.classList.add("selected"); var uInfoR = selections[userWhoSelectedR]; var uColorR = getColorForStudent(uInfoR.name || "學生", uInfoR.role); btn.style.borderColor = uColorR; btn.style.boxShadow = "0 0 0 3.5px " + uColorR + "55 inset"; } else if(!isMatched){ btn.style.boxShadow = ""; } if(isMatched || !interactive) btn.disabled = true; btn.onclick = function(){ if(matched[pid] || !interactive) return; selR = pid; var mySel = { left: selL, right: selR, name: myName, role: roomRole }; var newSelections = Object.assign({}, selections, { [myId]: mySel }); writeProgress(s.id, { selections: newSelections }); checkMatch(); }; rCol.appendChild(btn); }); function checkMatch(){ // 如果單邊已選，透過暫存變數保留，等待另一邊點擊 var currentSel = getProgress(s.id).selections || {}; var activeL = selL || Object.keys(currentSel).map(function(k){ return currentSel[k].left; }).filter(Boolean)[0]; var activeR = selR || Object.keys(currentSel).map(function(k){ return currentSel[k].right; }).filter(Boolean)[0]; if(activeL && activeR){ var newSelections = Object.assign({}, currentSel); delete newSelections[myId]; if(activeL === activeR){ playCorrectSound(); matched[activeL] = true; writeProgress(s.id, { matched: matched, lastWrong: null, selections: newSelections }); if(Object.keys(matched).length === pIds.length){ setTimeout(playVictorySound, 400); } } else { // 錯誤時強制變更為醒目的紅色對比邊框與背景 playWrongSound(); writeProgress(s.id, { lastWrong: { left: activeL, right: activeR, t: Date.now(), who: myName }, selections: newSelections }); } selL = null; selR = null; } } cols.appendChild(lCol); cols.appendChild(rCol); wrap.appendChild(cols); container.appendChild(wrap); var lastWrong = progress.lastWrong; if(lastWrong && lastWrong.t && (Date.now() - lastWrong.t < 1200)){ var wLeft = lCol.querySelector(`.match-item[data-id="${lastWrong.left}"]`); var wRight = rCol.querySelector(`.match-item[data-id="${lastWrong.right}"]`); if(wLeft && wRight){ wLeft.classList.add("wrong"); wRight.classList.add("wrong"); playWrongSound(); setTimeout(function(){ if(wLeft) wLeft.classList.remove("wrong", "selected"); if(wRight) wRight.classList.remove("wrong", "selected"); }, 600); } } function drawLines(){ var rect = wrap.getBoundingClientRect(); svg.setAttribute("viewBox","0 0 "+rect.width+" "+rect.height); svg.innerHTML = ""; Object.keys(matched).forEach(function(pid){ var l = lCol.querySelector(`.match-item[data-id="${pid}"]`); var r = rCol.querySelector(`.match-item[data-id="${pid}"]`); if(!l || !r) return; var lr = l.getBoundingClientRect(), rr = r.getBoundingClientRect(); var x1 = lr.right - rect.left, y1 = lr.top + lr.height/2 - rect.top; var x2 = rr.left - rect.left, y2 = rr.top + rr.height/2 - rect.top; var mx = (x1+x2)/2; var path = document.createElementNS("http://www.w3.org/2000/svg","path"); path.setAttribute("d", "M"+x1+","+y1+" C "+mx+","+y1+" "+mx+","+y2+" "+x2+","+y2); path.setAttribute("stroke", "var(--bamboo)"); path.setAttribute("stroke-width","3.5"); path.setAttribute("fill","none"); path.setAttribute("stroke-linecap","round"); svg.appendChild(path); }); } requestAnimationFrame(drawLines); window.addEventListener("resize", drawLines); }
